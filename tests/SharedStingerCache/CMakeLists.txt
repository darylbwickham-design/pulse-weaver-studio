cmake_minimum_required(VERSION 3.28)
project(PulseSharedStingerCacheTests LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
get_filename_component(REPO_ROOT "../.." ABSOLUTE BASE_DIR "${CMAKE_CURRENT_SOURCE_DIR}")
set(OBS_ROOT "${REPO_ROOT}/engine/obs-studio")
set(OBS_BUILD "${OBS_ROOT}/build_unstable" CACHE PATH "OBS build")
set(OBS_DEPS "${OBS_ROOT}/.deps/obs-deps-2026-07-15-x64" CACHE PATH "OBS dependencies")
set(MEDIA "${OBS_ROOT}/shared/media-playback/media-playback")
add_executable(PulseSharedStingerCacheTests main.cpp "${MEDIA}/cache.c" "${MEDIA}/decode.c" "${MEDIA}/media.c" "${MEDIA}/media-playback.c")
target_include_directories(PulseSharedStingerCacheTests PRIVATE "${OBS_ROOT}/libobs" "${OBS_BUILD}/config" "${OBS_DEPS}/include" "${MEDIA}" "${OBS_ROOT}/deps/w32-pthreads")
target_link_libraries(PulseSharedStingerCacheTests PRIVATE "${OBS_BUILD}/libobs/RelWithDebInfo/obs.lib")
foreach(lib avcodec avdevice avutil avformat swscale)
  target_link_libraries(PulseSharedStingerCacheTests PRIVATE "${OBS_DEPS}/lib/${lib}.lib")
endforeach()
if(WIN32)
  target_link_libraries(PulseSharedStingerCacheTests PRIVATE "${OBS_BUILD}/deps/w32-pthreads/RelWithDebInfo/w32-pthreads.lib")
endif()
