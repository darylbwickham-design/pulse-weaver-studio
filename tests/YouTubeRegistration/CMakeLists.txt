cmake_minimum_required(VERSION 3.28)
project(PulseYouTubeRegistrationTests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(Qt6 REQUIRED COMPONENTS Core Widgets)
add_executable(PulseYouTubeRegistrationTests main.cpp)
target_link_libraries(PulseYouTubeRegistrationTests PRIVATE Qt6::Core)
add_executable(PulseYouTubeRegistrationExport export.cpp)
target_link_libraries(PulseYouTubeRegistrationExport PRIVATE Qt6::Core)
add_executable(PulseYouTubeApplicationUiTests ui.cpp)
target_link_libraries(PulseYouTubeApplicationUiTests PRIVATE Qt6::Widgets)
if(MSVC)
  target_compile_options(PulseYouTubeRegistrationTests PRIVATE /utf-8)
  target_compile_options(PulseYouTubeApplicationUiTests PRIVATE /utf-8)
endif()
enable_testing()
add_test(NAME YouTubeRegistration COMMAND PulseYouTubeRegistrationTests)
add_test(NAME YouTubeApplicationUi COMMAND PulseYouTubeApplicationUiTests -platform minimal)
