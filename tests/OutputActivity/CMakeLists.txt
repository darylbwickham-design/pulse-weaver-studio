cmake_minimum_required(VERSION 3.28)
project(PulseOutputActivityTests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
get_filename_component(REPO_ROOT "../.." ABSOLUTE BASE_DIR "${CMAKE_CURRENT_SOURCE_DIR}")
set(OBS_ROOT "${REPO_ROOT}/engine/obs-studio")
set(OBS_BUILD "${OBS_ROOT}/build_unstable" CACHE PATH "OBS build used by the local runtime")
add_executable(PulseOutputActivityTests main.cpp)
target_include_directories(PulseOutputActivityTests PRIVATE "${OBS_ROOT}/libobs" "${OBS_BUILD}/config")
target_link_libraries(PulseOutputActivityTests PRIVATE "${OBS_BUILD}/libobs/RelWithDebInfo/obs.lib")
enable_testing()
add_test(NAME OutputActivity COMMAND PulseOutputActivityTests "${OBS_BUILD}/rundir/RelWithDebInfo")
