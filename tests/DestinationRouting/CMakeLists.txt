cmake_minimum_required(VERSION 3.28)
project(PulseDestinationRoutingTests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(Qt6 REQUIRED COMPONENTS Widgets)
set(frontend "${CMAKE_CURRENT_SOURCE_DIR}/../../engine/obs-studio/frontend/widgets/OBSBasic_PulseWeaver.cpp")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${frontend}")
file(READ "${frontend}" source)
string(FIND "${source}" "\t\tconnect(route, &QComboBox::currentIndexChanged, this, [this, route, key, label](int) {" begin)
string(FIND "${source}" "\n\t\tdestinations->addWidget(route);" end)
if(begin LESS 0 OR end LESS begin)
 message(FATAL_ERROR "Destination callback extraction anchors changed")
endif()
math(EXPR length "${end} - ${begin}")
string(SUBSTRING "${source}" ${begin} ${length} callback)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/callback-under-test.hpp" "${callback}")
add_executable(PulseDestinationRoutingTests main.cpp)
target_include_directories(PulseDestinationRoutingTests PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(PulseDestinationRoutingTests PRIVATE Qt6::Widgets)
add_custom_command(TARGET PulseDestinationRoutingTests POST_BUILD
 COMMAND ${CMAKE_COMMAND} -E copy_if_different "$<TARGET_FILE:Qt6::Core>" "$<TARGET_FILE:Qt6::Gui>" "$<TARGET_FILE:Qt6::Widgets>" "$<TARGET_FILE_DIR:PulseDestinationRoutingTests>"
 COMMAND ${CMAKE_COMMAND} -E make_directory "$<TARGET_FILE_DIR:PulseDestinationRoutingTests>/platforms"
 COMMAND ${CMAKE_COMMAND} -E copy_if_different "$<TARGET_FILE:Qt6::QMinimalIntegrationPlugin>" "$<TARGET_FILE_DIR:PulseDestinationRoutingTests>/platforms")
enable_testing()
add_test(NAME PulseDestinationRoutingTests COMMAND PulseDestinationRoutingTests)
