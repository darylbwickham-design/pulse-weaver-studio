cmake_minimum_required(VERSION 3.28)
project(PulseStingerFramePoolTests LANGUAGES CXX)
get_filename_component(REPO_ROOT "../.." ABSOLUTE BASE_DIR "${CMAKE_CURRENT_SOURCE_DIR}")
set(OBS_ROOT "${REPO_ROOT}/engine/obs-studio")
set(OBS_BUILD "${OBS_ROOT}/build_unstable" CACHE PATH "Existing OBS build")
add_executable(PulseStingerFramePoolTests main.cpp)
target_compile_features(PulseStingerFramePoolTests PRIVATE cxx_std_17)
target_include_directories(PulseStingerFramePoolTests PRIVATE "${OBS_ROOT}/libobs" "${OBS_BUILD}/config")
target_link_libraries(PulseStingerFramePoolTests PRIVATE "${OBS_BUILD}/libobs/RelWithDebInfo/obs.lib")
enable_testing()
add_test(NAME PulseStingerFramePoolTests COMMAND PulseStingerFramePoolTests)
