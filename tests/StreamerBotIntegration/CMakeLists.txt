cmake_minimum_required(VERSION 3.28)
project(PulseBasicControlsTests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Widgets)
add_executable(PulseBasicControlsTests main.cpp)
target_compile_options(PulseBasicControlsTests PRIVATE /utf-8)
target_link_libraries(PulseBasicControlsTests PRIVATE Qt6::Widgets)
enable_testing()
add_test(NAME BasicControls COMMAND PulseBasicControlsTests)
set_tests_properties(BasicControls PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=minimal")
