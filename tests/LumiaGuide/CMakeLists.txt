cmake_minimum_required(VERSION 3.28)
project(PulseLumiaGuideChecks LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(Qt6 REQUIRED COMPONENTS Widgets)
add_executable(PulseLumiaGuideChecks main.cpp)
target_compile_options(PulseLumiaGuideChecks PRIVATE /utf-8)
target_link_libraries(PulseLumiaGuideChecks PRIVATE Qt6::Widgets)
enable_testing()
add_test(NAME LumiaGuide COMMAND PulseLumiaGuideChecks "${CMAKE_CURRENT_BINARY_DIR}/screenshots")
