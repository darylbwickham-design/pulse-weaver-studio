cmake_minimum_required(VERSION 3.28)
project(PulseKeypadTests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(Qt6 REQUIRED COMPONENTS Widgets Network)
get_filename_component(REPO "../.." ABSOLUTE BASE_DIR "${CMAKE_CURRENT_SOURCE_DIR}")
set(OBS_ROOT "${REPO}/engine/obs-studio")
set(OBS_BUILD "" CACHE PATH "Existing compatible native build")
add_executable(PulseKeypadTests main.cpp)
target_compile_options(PulseKeypadTests PRIVATE /utf-8 /wd4996)
target_compile_definitions(PulseKeypadTests PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_include_directories(PulseKeypadTests PRIVATE "${OBS_ROOT}/libobs" "${OBS_BUILD}/config" "${OBS_ROOT}/frontend/api")
target_link_libraries(PulseKeypadTests PRIVATE Qt6::Widgets Qt6::Network "${OBS_BUILD}/libobs/RelWithDebInfo/obs.lib" "${OBS_BUILD}/frontend/api/RelWithDebInfo/obs-frontend-api.lib")
