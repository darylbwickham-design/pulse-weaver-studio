cmake_minimum_required(VERSION 3.28)
if(NOT TARGET OBS::libobs)
  project(altron_teleport VERSION 0.1.0 LANGUAGES CXX)
endif()
set(CMAKE_CXX_STANDARD 17)
if(TARGET OBS::libobs)
  add_library(altron-teleport MODULE teleport.cpp)
  target_link_libraries(altron-teleport PRIVATE OBS::libobs)
  set_target_properties_obs(altron-teleport PROPERTIES PREFIX "" FOLDER plugins/altron-teleport)
else()
  if(NOT OBS_SOURCE_DIR OR NOT OBS_BUILD_DIR)
    message(FATAL_ERROR "Set OBS_SOURCE_DIR and OBS_BUILD_DIR to the matching Pulse Weaver/OBS SDK")
  endif()
  add_library(altron-teleport MODULE teleport.cpp)
  target_include_directories(altron-teleport PRIVATE "${OBS_SOURCE_DIR}/libobs" "${OBS_BUILD_DIR}/config")
  target_link_libraries(altron-teleport PRIVATE "${OBS_BUILD_DIR}/libobs/RelWithDebInfo/obs.lib")
  set_target_properties(altron-teleport PROPERTIES PREFIX "")
  install(TARGETS altron-teleport RUNTIME DESTINATION obs-plugins/64bit LIBRARY DESTINATION obs-plugins/64bit)
  install(FILES data/white-teleport.effect DESTINATION data/obs-plugins/altron-teleport)
  if(ALTRON_BUILD_TESTS)
    add_executable(altron-native-test test-native.cpp)
    target_include_directories(altron-native-test PRIVATE "${OBS_SOURCE_DIR}/libobs" "${OBS_BUILD_DIR}/config")
    target_link_libraries(altron-native-test PRIVATE "${OBS_BUILD_DIR}/libobs/RelWithDebInfo/obs.lib")
  endif()
endif()
