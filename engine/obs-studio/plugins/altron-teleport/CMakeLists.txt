cmake_minimum_required(VERSION 3.28)
if(NOT TARGET OBS::libobs)
  project(altron_teleport VERSION 0.2.0 LANGUAGES CXX)
endif()
set(CMAKE_CXX_STANDARD 17)
find_package(Qt6 REQUIRED COMPONENTS Widgets Network)
if(TARGET OBS::libobs)
  add_library(altron-teleport MODULE teleport.cpp native-control.cpp)
  target_link_libraries(altron-teleport PRIVATE OBS::libobs OBS::frontend-api Qt::Widgets Qt::Network)
  set_target_properties_obs(altron-teleport PROPERTIES PREFIX "" FOLDER plugins/altron-teleport)
else()
  if(NOT OBS_SOURCE_DIR OR NOT OBS_BUILD_DIR)
    message(FATAL_ERROR "Set OBS_SOURCE_DIR and OBS_BUILD_DIR to the matching Pulse Weaver/OBS SDK")
  endif()
  add_library(altron-teleport MODULE teleport.cpp native-control.cpp)
  target_include_directories(altron-teleport PRIVATE "${OBS_SOURCE_DIR}/libobs" "${OBS_SOURCE_DIR}/frontend/api" "${OBS_BUILD_DIR}/config")
  target_link_libraries(altron-teleport PRIVATE "${OBS_BUILD_DIR}/libobs/RelWithDebInfo/obs.lib" "${OBS_BUILD_DIR}/frontend/api/RelWithDebInfo/obs-frontend-api.lib" Qt6::Widgets Qt6::Network)
  set_target_properties(altron-teleport PROPERTIES PREFIX "")
  install(TARGETS altron-teleport RUNTIME DESTINATION obs-plugins/64bit LIBRARY DESTINATION obs-plugins/64bit)
  install(FILES data/white-teleport.effect DESTINATION data/obs-plugins/altron-teleport)
  if(ALTRON_BUILD_TESTS)
    set(ALTRON_TEST_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}")
    if(NOT EXISTS "${ALTRON_TEST_SOURCE_DIR}/test-native.cpp")
      set(ALTRON_TEST_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../../../tests/AltronTeleport")
    endif()
    add_executable(altron-native-test "${ALTRON_TEST_SOURCE_DIR}/test-native.cpp")
    target_include_directories(altron-native-test PRIVATE "${OBS_SOURCE_DIR}/libobs" "${OBS_BUILD_DIR}/config")
    target_link_libraries(altron-native-test PRIVATE "${OBS_BUILD_DIR}/libobs/RelWithDebInfo/obs.lib")
    add_executable(altron-native-control-test "${ALTRON_TEST_SOURCE_DIR}/test-native-control.cpp" native-control.cpp)
    target_include_directories(altron-native-control-test PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}" "${OBS_SOURCE_DIR}/libobs" "${OBS_SOURCE_DIR}/frontend/api" "${OBS_BUILD_DIR}/config")
    target_link_libraries(altron-native-control-test PRIVATE "${OBS_BUILD_DIR}/libobs/RelWithDebInfo/obs.lib" "${OBS_BUILD_DIR}/frontend/api/RelWithDebInfo/obs-frontend-api.lib" Qt6::Widgets Qt6::Network)
  endif()
endif()
